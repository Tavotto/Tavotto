import hashlib,json,pathlib,shutil,subprocess

out=pathlib.Path('/tmp/cla-timing-independent-review-830bd89c')
snapshot=out/'snapshot'; source=snapshot/'scripts/timing-plan.mjs'; commands=[]
def run(argv,cwd,log):
    r=subprocess.run(argv,cwd=cwd,text=True,capture_output=True,timeout=15)
    (out/log).write_text(r.stdout+r.stderr)
    commands.append({'argv':argv,'cwd':str(cwd),'exit':r.returncode,'log':log})
    return r
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()

# Restore the exact initial source from the two independently observed edits;
# assert its full hash before attributing an old-source failure to that snapshot.
initial=out/'initial-snapshot';shutil.copytree(snapshot,initial,dirs_exist_ok=True)
old=(initial/'scripts/timing-plan.mjs').read_text().replace('if(readyAfterEvent<start)','if(readyAfterEvent<=start)')
cli_new="""    fd=fs.openSync(process.argv[2],'r');const stat=fs.fstatSync(fd);
    if(!stat.isFile())throw Error('regular_input_file_required');if(stat.size>65536)throw Error('input_too_large');
    const buffer=Buffer.alloc(65537);let length=0,count;
    do{count=fs.readSync(fd,buffer,length,buffer.length-length,null);length+=count;}while(count>0 && length<buffer.length);
    if(length>65536)throw Error('input_too_large');"""
cli_old="""    fd=fs.openSync(process.argv[2],'r');if(!fs.fstatSync(fd).isFile())throw Error('regular_input_file_required');
    const buffer=Buffer.alloc(65537),length=fs.readSync(fd,buffer,0,buffer.length,null);if(length>65536)throw Error('input_too_large');"""
assert old.count(cli_new)==1
old=old.replace(cli_new,cli_old);(initial/'scripts/timing-plan.mjs').write_text(old)
assert sha(initial/'scripts/timing-plan.mjs')=='0b562df2210853c1bd89d5e032fd92830da3c045d8f1ed4299f3111a832af22a'
red=run(['node','--test','tests/timing-independent.test.mjs'],initial,'independent-red-original-source.txt')
green=run(['node','--test','tests/timing-independent.test.mjs'],snapshot,'independent-green-final-source.txt')
assert red.returncode==1 and green.returncode==0

    mutant=out/'mutants'/name;shutil.copytree(snapshot,mutant,dirs_exist_ok=True)
