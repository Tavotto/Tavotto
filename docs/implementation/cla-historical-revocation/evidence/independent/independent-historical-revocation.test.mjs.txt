const snap=f=>Object.fromEntries(['state','delegated_grants','delegated_controls','delegated_receipts','packets','signatures','eligibility_blocks','outbox','head_requests'].map(t=>[t,f.sql.prepare('SELECT * FROM cla_native_v2_'+t+' ORDER BY rowid').all().map(r=>({...r}))]));
function rotated(f,change={}){const c={...structuredClone(f.c),holderAuthorizationId:randomToken(),holderAuthorizationKey:randomToken(),signingEnabled:false,delegatedCountersigningEnabled:false,...change};return new LiveService(c,new LiveStore(f.db,undefined,c),f.gh,f.key);}
async function denied(f,fn,code){const before=snap(f);await assert.rejects(fn,code?e=>e.code===code:undefined);assert.deepEqual(snap(f),before);}
test('independent historical old key revocation after rotation preserves all personal evidence',()=>iso(async f=>{
 await f.install();const p=await f.create();await f.sign(p);const s=rotated(f),before=snap(f),rev=await f.store.revision(),record=await f.control('revoke');
 await assert.rejects(s.delegation.append(record));await s.delegation.appendHistoricalRevocation(f.grant.id,record);
 const after=snap(f);assert.equal(after.delegated_controls.length,before.delegated_controls.length+1);assert.equal(after.delegated_controls.at(-1).action,'revoke');assert.equal(await f.store.revision(),rev+1);
 for(const t of ['delegated_grants','delegated_receipts','packets','signatures','eligibility_blocks'])assert.deepEqual(after[t],before[t]);
 await assert.rejects(f.s.delegation.qualified(p,(await f.store.signatures(p.id))[0],f.now()));
 const a=await s.audit(f.args('contributor')[0],p.id);assert.equal(a.delegated.current,false);assert.equal(JSON.parse(a.delegated.controls.at(-1).body).action,'revoke');
}));
