function snapshot(db){

 const rows=objects.filter(o=>o.type==='table').map(o=>{const columns=db.prepare(`PRAGMA table_xinfo(${q(o.name)})`).all().map(c=>c.name);
 const select=columns.map(c=>`typeof(${q(c)}) AS ${q('type_'+c)},hex(${q(c)}) AS ${q('hex_'+c)},quote(${q(c)}) AS ${q('quote_'+c)}`);
 const stmt=db.prepare(`SELECT rowid AS _rowid,${select.join(',')} FROM ${q(o.name)} ORDER BY rowid`);stmt.setReadBigInts(true);
 return [o.name,stmt.all()];});
 return JSON.stringify({objects,rows},(_,v)=>typeof v==='bigint'?v.toString():v);
}
